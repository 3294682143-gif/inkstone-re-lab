async function stopAndDrainHeadlessTurn(host, chatId, requestId) {
  await host.stopTurn(requestId);
  const deadline = host.now() + STOP_DRAIN_TIMEOUT_MS;
  while (host.now() < deadline) {
    const history = await host.history(chatId);
    if (history.activeRequestId !== requestId) return true;
    await host.sleep(POLL_INTERVAL_MS);
  }
  return false;
}
