  projectReactions(chatId, history) {
    const reactions = this.reactionsByChat.get(chatId);
    if (!reactions?.size) return history;
    return {
      ...history,
      messages: history.messages.map((message) => {
        if (message.role !== "assistant" || !message.turnId || !reactions.has(message.turnId)) {
          return message;
        }
        return {
          ...message,
          reaction: reactions.get(message.turnId)
        };
      })
    };
  }
