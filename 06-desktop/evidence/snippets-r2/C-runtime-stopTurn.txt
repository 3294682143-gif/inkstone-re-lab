  async stopTurn(requestId) {
    this.retireExecution(requestId);
    const ready = await this.requireReady();
    const active = ready.projection.activeTurn(requestId);
    if (!active) {
      this.terminalFailureObservers.delete(requestId);
      this.clearTurnStartupWatchdog(requestId);
      this.activeTurnSessions.delete(requestId);
      this.retireExecution(requestId);
      return;
    }
    this.clearTurnStartupWatchdog(requestId, active.turnId);
    const result = await ready.supervisor.callRuntime("turn.stop", {
      requestId,
      expectedTurnId: active.turnId
    });
    if (result["stopped"] !== true) {
      throw new Error("Core Runtime did not stop the turn");
    }
    this.assertCurrent(ready);
    if (ready.projection.activeTurn(requestId)?.turnId !== active.turnId) {
      return;
    }
    this.stalledTurnFailures.delete(requestId);
    this.retireTurnApprovals(requestId, active.turnId);
    this.retireHarnessQuestion(requestId, active.turnId);
    this.firstTurnTitleRequests.delete(requestId);
    this.closeObservedToolCalls(active.turnId, active.chatId);
    ready.gateway.releaseExternalSkills(requestId);
    ready.projection.stopTurn(requestId);
    this.terminalFailureObservers.delete(requestId);
    this.activeTurnSessions.delete(requestId);
    this.retireExecution(requestId);
  }
