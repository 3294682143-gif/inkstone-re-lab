  const driver = new HeadlessChatDriver(host, write);
  driver.ready();
  try {
    for await (const line of lines) {
      if (driver.isShutdownRequested) break;
      if (!line.trim()) continue;
      await driver.handleLine(line);
      if (driver.isShutdownRequested) break;
    }
  } finally {
    await driver.shutdown();
  }
}
function parseRequest(value) {
  if (!isRecord(value)) {
    throw headlessError("HEADLESS_INVALID_REQUEST", "Headless request must be an object");
  }
  exactKeys(value, [
    "id",
    "method",
    "params"
  ]);
  const id2 = requiredString(value["id"], "id");
  const method = requiredString(value["method"], "method");
  if (id2.length > 128 || method.length > 128) {
    throw headlessError("HEADLESS_INVALID_REQUEST", "Headless request identifier is too long");
  }
  const params = value["params"] ?? {};
  if (!isRecord(params)) {
    throw headlessError("HEADLESS_INVALID_PARAMS", "params must be an object");
  }
  return {
    id: id2,
    method,
    params
  };
}
