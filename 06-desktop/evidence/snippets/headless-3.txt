      await runHeadlessChatDriver(lines, host, (output) => writeJsonLine(io.stdout, output));
    } finally {
      lines.close();
    }
    return 0;
  } catch (cause) {
    writeJsonLine(io.stderr, {
      type: "error",
      ...safeHeadlessError(cause)
    });
    return 1;
  }
}
